(function () {
    const CADEN_VERSION = "0.6.30";
    const CADEN_CONFIG = {
        appId: "RCM",
        appVersion: (typeof APP_BUILD_META !== 'undefined' && APP_BUILD_META.appVersion) ? APP_BUILD_META.appVersion : (typeof APP_VERSION !== 'undefined' ? APP_VERSION : 'Unknown'),
        serverUrl: "https://caden.prasanti.com",
        wssUrl: "wss://caden.prasanti.com/v1/stream",
        trialDays: 30,
        logLevel: 2
    };
    const KEYS = {
        token: 'RCM_AUTH_TOKEN',
        userInfo: 'RCM_USER_INFO',
        genesis: 'RCM_GENESIS',
        pendingReg: 'RCM_PENDING_REG'
    };

    function generateDeviceFingerprint() {
        try {
            const parts = [];
            parts.push('W:' + screen.width);
            parts.push('H:' + screen.height);
            parts.push('CPU:' + (navigator.hardwareConcurrency || 0));
            parts.push('RAM:' + (navigator.deviceMemory || 0));
            parts.push('TZ:' + (Intl.DateTimeFormat().resolvedOptions().timeZone || ''));
            try {
                const c = document.createElement('canvas');
                const gl = c.getContext('webgl') || c.getContext('experimental-webgl');
                if (gl) { const d = gl.getExtension('WEBGL_debug_renderer_info'); if (d) parts.push('GPU:' + gl.getParameter(d.UNMASKED_RENDERER_WEBGL)); }
            } catch(e) {}
            const combined = parts.join('|'); let hash = 0;
            for (let i = 0; i < combined.length; i++) { hash = ((hash << 5) - hash) + combined.charCodeAt(i); hash |= 0; }
            const prefix = CADEN_CONFIG.appId.split('_')[0].toUpperCase();
            return prefix + "-" + Math.abs(hash).toString(16).toUpperCase().padStart(8, '0');
        } catch (e) { return "RCM-GENERIC-00000000"; }
    }

    function collectSystemInfo() {
        const info = {
            userAgent: navigator.userAgent,
            platform: navigator.platform,
            cpuCores: navigator.hardwareConcurrency || 0,
            deviceMemory: navigator.deviceMemory ? navigator.deviceMemory + ' GB' : 'Unknown',
            screenRes: screen.width + 'x' + screen.height,
            timezone: (Intl.DateTimeFormat().resolvedOptions().timeZone || ''),
            webglRenderer: (function(){ try { const c=document.createElement('canvas'); const gl=c.getContext('webgl')||c.getContext('experimental-webgl'); const d=gl&&gl.getExtension('WEBGL_debug_renderer_info'); return d?gl.getParameter(d.UNMASKED_RENDERER_WEBGL):''; } catch(e){ return ''; } })(),
            appVersion: CADEN_CONFIG.appVersion,
            cadenVersion: CADEN_VERSION
        };
        if (typeof APP_BUILD_META !== 'undefined') {
            info.buildDataVersion = APP_BUILD_META.buildDataVersion || 'Unknown';
            info.embedDataVersion = APP_BUILD_META.embedDataVersion || 'Unknown';
            info.sourceExcelFile = APP_BUILD_META.sourceExcelFile || 'Unknown';
        }
        return info;
    }

    function syncProfile() {
        try {
            const payload = { deviceId: generateDeviceFingerprint(), systemInfo: collectSystemInfo() };
            fetch(CADEN_CONFIG.serverUrl + '/v1/profile', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) }).catch(function(){});
        } catch(e) {}
    }

    function executeLockdownScreen(reason) {
        let overlay = document.getElementById('caden-lockdown-overlay');
        if (!overlay) {
            overlay = document.createElement('div');
            overlay.id = 'caden-lockdown-overlay';
            overlay.style.cssText = 'position:fixed;top:0;left:0;width:100vw;height:100vh;background:#0f172a;z-index:999999;display:flex;align-items:center;justify-content:center;flex-direction:column;color:#fff;font-family:sans-serif;';
            document.body.appendChild(overlay);
        }
        overlay.innerHTML = '<div style="background:#1e293b;padding:2rem;border-radius:8px;text-align:center;max-width:400px;"><h2 style="color:#ef4444;margin-bottom:1rem;">CADEN Governance Lockdown</h2><p style="margin-bottom:1rem;">' + reason + '</p><p style="font-size:0.8rem;color:#94a3b8;">Device ID: ' + generateDeviceFingerprint() + ' | Script v' + CADEN_VERSION + '</p></div>';
        document.body.querySelectorAll(':not(#caden-lockdown-overlay)').forEach(function(el){ if(el.parentNode) el.style.display='none'; });
        document.body.appendChild(overlay);
    }

    function revealApp() {
        const modal = document.getElementById('caden-reg-modal') || document.querySelector('.caden-modal');
        if (modal) modal.style.display = 'none';
        window.dispatchEvent(new Event('CADEN_READY'));
    }

    function showRevokedAppealForm() {
        if (document.getElementById('caden-appeal-overlay')) return;
        const appealHTML = `
            <div id="caden-appeal-overlay" style="position:fixed;top:0;left:0;width:100vw;height:100vh;background:rgba(0,0,0,0.95);color:#fff;display:flex;align-items:center;justify-content:center;z-index:999999;font-family:sans-serif;">
                <div style="background:#1e293b;padding:2rem;border-radius:8px;width:90%;max-width:400px;box-shadow:0 10px 25px rgba(0,0,0,0.5);">
                    <h2 style="color:#ef4444;margin-top:0;text-align:center;">License Revoked</h2>
                    <p style="text-align:center;font-size:0.9rem;color:#94a3b8;margin-bottom:1.5rem;">Please submit an appeal to restore access.</p>
                    <form id="caden-appeal-form">
                        <input type="text" id="appeal-name" placeholder="Full Name" required style="width:100%;padding:0.75rem;margin-bottom:1rem;background:#0f172a;border:1px solid #334155;color:#fff;border-radius:4px;box-sizing:border-box;">
                        <input type="email" id="appeal-email" placeholder="Email Address" required style="width:100%;padding:0.75rem;margin-bottom:1rem;background:#0f172a;border:1px solid #334155;color:#fff;border-radius:4px;box-sizing:border-box;">
                        <input type="text" id="appeal-wa" placeholder="WhatsApp Number" required style="width:100%;padding:0.75rem;margin-bottom:1.5rem;background:#0f172a;border:1px solid #334155;color:#fff;border-radius:4px;box-sizing:border-box;">
                        <button type="submit" style="width:100%;padding:0.75rem;background:#3b82f6;color:#fff;border:none;border-radius:4px;font-weight:bold;cursor:pointer;">Submit Appeal</button>
                    </form>
                    <div id="appeal-success" style="display:none;text-align:center;color:#10b981;margin-top:1rem;">Appeal submitted successfully. Please wait for approval.</div>
                </div>
            </div>`;
        document.body.insertAdjacentHTML('beforeend', appealHTML);
        document.getElementById('caden-appeal-form').addEventListener('submit', async function(e) {
            e.preventDefault();
            const name = document.getElementById('appeal-name').value;
            const email = document.getElementById('appeal-email').value;
            const wa = document.getElementById('appeal-wa').value;
            try {
                const res = await fetch(CADEN_CONFIG.serverUrl + '/v1/appeal?deviceId=' + generateDeviceFingerprint() + '&ownerName=' + encodeURIComponent(name) + '&ownerEmail=' + encodeURIComponent(email) + '&ownerWhatsapp=' + encodeURIComponent(wa), { method: 'POST' });
                if (res.ok) {
                    document.getElementById('caden-appeal-form').style.display = 'none';
                    document.getElementById('appeal-success').style.display = 'block';
                } else {
                    alert('Failed to submit appeal. Please try again.');
                }
            } catch (err) {
                alert('Network error. Please check your connection.');
            }
        });
        console.warn('CADEN: License is REVOKED. Appeal form displayed.');
    }

    function showRegistrationModal() {
        let modal = document.getElementById('caden-reg-modal');
        if (!modal) {
            modal = document.createElement('div');
            modal.id = 'caden-reg-modal';
            modal.style.cssText = 'position:fixed;top:0;left:0;width:100vw;height:100vh;background:rgba(0,0,0,0.8);z-index:999998;display:flex;align-items:center;justify-content:center;';
            document.body.appendChild(modal);
        }
        modal.innerHTML = '<div style="background:#fff;padding:2rem;border-radius:8px;width:300px;color:#000;"><h3 style="margin-top:0;">CADEN Registration</h3><input id="caden-reg-name" placeholder="Full Name" style="width:100%;margin-bottom:10px;padding:8px;box-sizing:border-box;"><input id="caden-reg-email" placeholder="Email" style="width:100%;margin-bottom:10px;padding:8px;box-sizing:border-box;"><input id="caden-reg-wa" placeholder="WhatsApp" style="width:100%;margin-bottom:10px;padding:8px;box-sizing:border-box;"><button id="caden-reg-submit" style="width:100%;padding:10px;background:#2563eb;color:#fff;border:none;border-radius:4px;cursor:pointer;">Activate</button></div>';
        modal.style.display = 'flex';
        document.getElementById('caden-reg-submit').onclick = async function () {
            const name = document.getElementById('caden-reg-name').value;
            const email = document.getElementById('caden-reg-email').value;
            const wa = document.getElementById('caden-reg-wa').value;
            if (!name) return alert('Name required');
            if (!email || !email.includes('@')) return alert('Valid email required');
            const devId = generateDeviceFingerprint();
            const localExpiry = new Date(Date.now() + 30 * 24 * 60 * 60 * 1000).toISOString();
            const userInfoObj = { name: name, email: email, whatsapp: wa, expires_at: localExpiry };
            console.log('CADEN: Saving initial userInfo + local 30-day expiry to localStorage:', userInfoObj);
            localStorage.setItem(KEYS.userInfo, JSON.stringify(userInfoObj));
            
            const url = CADEN_CONFIG.serverUrl + '/v1/activate?appId=' + CADEN_CONFIG.appId + '&deviceId=' + devId + '&ownerName=' + encodeURIComponent(name) + '&ownerEmail=' + encodeURIComponent(email) + '&ownerWhatsapp=' + encodeURIComponent(wa);
            try {
                const res = await fetch(url, { method: 'POST' });
                const data = await res.json();
                if (res.ok && data.token) {
                    localStorage.setItem(KEYS.token, data.token);
                    localStorage.removeItem(KEYS.pendingReg);
                    if (data.owner_name) userInfoObj.name = data.owner_name;
                    if (data.owner_email) userInfoObj.email = data.owner_email;
                    if (data.owner_whatsapp) userInfoObj.whatsapp = data.owner_whatsapp;
                    if (data.expires_at) userInfoObj.expires_at = data.expires_at;
                    localStorage.setItem(KEYS.userInfo, JSON.stringify(userInfoObj));
                    syncProfile();
                    modal.remove();
                    if (data.status === 'REVOKED') {
                        showRevokedAppealForm();
                        return;
                    }
                    revealApp();
                } else {
                    console.warn('CADEN: server rejected registration, proceeding offline.');
                    localStorage.setItem(KEYS.pendingReg, JSON.stringify({ url: url, method: 'POST' }));
                    modal.remove();
                    revealApp();
                }
            } catch (e) {
                console.warn('CADEN: network error, proceeding offline.');
                localStorage.setItem(KEYS.pendingReg, JSON.stringify({ url: url, method: 'POST' }));
                modal.remove();
                revealApp();
            }
        };
    }

    async function autoActivate() {
        const token = localStorage.getItem(KEYS.token);
        const devId = generateDeviceFingerprint();
        const userInfo = JSON.parse(localStorage.getItem(KEYS.userInfo) || '{}');
        const url = CADEN_CONFIG.serverUrl + '/v1/activate?appId=' + CADEN_CONFIG.appId + '&deviceId=' + devId + '&ownerName=' + encodeURIComponent(userInfo.name||'') + '&ownerEmail=' + encodeURIComponent(userInfo.email||'') + '&ownerWhatsapp=' + encodeURIComponent(userInfo.whatsapp||'');
        try {
            const res = await fetch(url, { method: 'POST' });
            const data = await res.json();
            if (!res.ok) { 
                if (data.detail === 'Device Revoked') executeLockdownScreen('License has been revoked by administrator.'); 
                return; 
            }
            const now = Math.floor(Date.now() / 1000);
            if (data.status === 'REVOKED') { 
                showRevokedAppealForm(); 
                return; 
            }
            if (data.expires_at) { 
                const expDate = new Date(data.expires_at).getTime() / 1000; 
                if (expDate < (now - 300)) { 
                    executeLockdownScreen('License has expired.'); 
                    return; 
                } 
            }
            if (data.owner_name || data.owner_email || data.owner_whatsapp || data.expires_at) {
                const ci = JSON.parse(localStorage.getItem(KEYS.userInfo) || '{}');
                if (data.owner_name) ci.name = data.owner_name;
                if (data.owner_email) ci.email = data.owner_email;
                if (data.owner_whatsapp) ci.whatsapp = data.owner_whatsapp;
                if (data.expires_at) ci.expires_at = data.expires_at;
                console.log('CADEN: Merging server userInfo into localStorage:', ci);
                localStorage.setItem(KEYS.userInfo, JSON.stringify(ci));
            }
            if (data.token) localStorage.setItem(KEYS.token, data.token);
            syncProfile();
            revealApp();
        } catch (e) { 
            console.error('CADEN activation error:', e); 
            if (token) revealApp(); 
        }
    }

    function attemptPendingRegistration() {
        const pending = localStorage.getItem(KEYS.pendingReg);
        if (!pending || !navigator.onLine) return;
        try {
            const req = JSON.parse(pending);
            fetch(req.url, { method: req.method }).then(function(res){ return res.json(); }).then(function(data){
                if (data.token) {
                    localStorage.setItem(KEYS.token, data.token);
                    localStorage.removeItem(KEYS.pendingReg);
                    if (data.owner_name || data.owner_email || data.owner_whatsapp || data.expires_at) {
                        const ci = JSON.parse(localStorage.getItem(KEYS.userInfo) || '{}');
                        if (data.owner_name) ci.name = data.owner_name;
                        if (data.owner_email) ci.email = data.owner_email;
                        if (data.owner_whatsapp) ci.whatsapp = data.owner_whatsapp;
                        if (data.expires_at) ci.expires_at = data.expires_at;
                        console.log('CADEN: Merging server userInfo after offline sync:', ci);
                        localStorage.setItem(KEYS.userInfo, JSON.stringify(ci));
                    }
                    syncProfile();
                    console.log('CADEN: offline registration synced successfully.');
                    autoActivate();
                }
            }).catch(function(e){ console.log('CADEN: pending registration retry failed.', e); });
        } catch(e) {}
    }

    window.addEventListener('online', function(){ attemptPendingRegistration(); autoActivate(); });
    setInterval(function(){ if (navigator.onLine) attemptPendingRegistration(); }, 5 * 60 * 1000);

    function init() {
        const badge = document.createElement('div');
        badge.style.cssText = 'position:fixed;top:10px;right:10px;background:#1e293b;color:#fff;padding:5px 10px;border-radius:4px;font-size:10px;z-index:999999;font-family:sans-serif;';
        badge.innerHTML = '🛡️ Protected by CADEN<br>ID: ' + generateDeviceFingerprint();
        document.body.appendChild(badge);
        const token = localStorage.getItem(KEYS.token);
        const hasGenesis = localStorage.getItem(KEYS.genesis);
        const hasUserInfo = localStorage.getItem(KEYS.userInfo);
        if (!token && !hasGenesis && !hasUserInfo) {
            showRegistrationModal();
        } else {
            autoActivate();
        }
        if (navigator.onLine) {
            syncProfile();
            attemptPendingRegistration();
        }
    }

    if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', init); else init();

    window.storeSystemTelemetry = function(type, payload, level) {
        if (!navigator.onLine) return;
        const devId = generateDeviceFingerprint();
        fetch(CADEN_CONFIG.serverUrl + '/v1/telemetry', {
            method: 'POST',
            headers: {'Content-Type': 'application/json'},
            body: JSON.stringify({
                appId: CADEN_CONFIG.appId,
                entries: [{
                    epoch: Math.floor(Date.now()/1000),
                    level: level || 2,
                    type: type,
                    detail: typeof payload === 'string' ? payload : JSON.stringify(payload),
                    device: devId
                }]
            })
        }).catch(()=>{});
    };

    window.logCadenOperation = function(type, msg){ console.log('CADEN Log:', type, msg); };

    try { window.syncProfile = syncProfile; } catch(e){}
    try { window.generateDeviceFingerprint = generateDeviceFingerprint; } catch(e){}
    try { window.updateDashboardFooter = updateDashboardFooter; } catch(e){}
    try { window.CADEN_CONFIG = CADEN_CONFIG; } catch(e){}
    try { window.KEYS = KEYS; } catch(e){}

    let lastActivitySync = Date.now();
    function handleUserActivity() {
        const now = Date.now();
        if (now - lastActivitySync > 60000) {
            if (navigator.onLine) syncProfile();
            lastActivitySync = now;
        }
    }
    ['keydown', 'mousemove', 'click', 'scroll'].forEach(evt => window.addEventListener(evt, handleUserActivity));

    setInterval(async function () {
        try {
            if (!navigator.onLine) return;
            const fp = generateDeviceFingerprint();
            const r = await fetch(CADEN_CONFIG.serverUrl + '/v1/activate?appId=' + encodeURIComponent(CADEN_CONFIG.appId) + '&deviceId=' + encodeURIComponent(fp), {method:'POST'});
            if (!r.ok) return;
            const j = await r.json();
            if (j && (j.owner_name || j.owner_email || j.owner_whatsapp || j.expires_at)) {
                const cur = JSON.parse(localStorage.getItem(KEYS.userInfo) || '{}');
                if (j.owner_name) cur.name = j.owner_name;
                if (j.owner_email) cur.email = j.owner_email;
                if (j.owner_whatsapp) cur.whatsapp = j.owner_whatsapp;
                if (j.expires_at) cur.expires_at = j.expires_at;
                localStorage.setItem(KEYS.userInfo, JSON.stringify(cur));
                if (typeof updateDashboardFooter === 'function') { try { updateDashboardFooter(cur); } catch (e) {} }
            }
        } catch (e) {}
    }, 60000);

    let lastNavLog = 0;
    document.addEventListener('click', function (e) {
        try {
            const now = Date.now(); if (now - lastNavLog < 3000) return; lastNavLog = now;
            const t = ((e.target && (e.target.innerText || e.target.getAttribute('aria-label') || e.target.id || e.target.tagName)) || '').toString().trim().slice(0, 40) || '(unknown)';
            storeSystemTelemetry('NAVIGATION', t, 2);
        } catch (err) {}
    });

    window.getCadenLicenseStatus = function() {
        const token = localStorage.getItem(KEYS.token);
        if(!token) return 'UNREGISTERED';
        try {
            const payload = JSON.parse(atob(token.split('.')[1]));
            const exp = payload.exp * 1000;
            if (Date.now() > exp) return 'EXPIRED';
            return 'ACTIVE';
        } catch(e) { return 'ERROR'; }
    };
})();
