<!-- START CADEN LICENSING & GOVERNANCE COMPONENT (v0.6.3) -->
<script>
(function () {
    const CADEN_VERSION = "0.6.5";
    const CADEN_CONFIG = {
        appId: "RCM",
        appVersion: typeof APP_VERSION !== 'undefined' ? APP_VERSION : 'Unknown',
        serverUrl: "https://caden.prasanti.com",
        wssUrl: "wss://caden.prasanti.com/v1/stream",
        trialDays: 3, 
        logLevel: 2
    };

    const KEYS = {
        token: `${CADEN_CONFIG.appId}_AUTH_TOKEN`,
        timeline: `${CADEN_CONFIG.appId}_LAST_SEEN`,
        genesis: `${CADEN_CONFIG.appId}_GENESIS_BUNCH`,
        logs: `${CADEN_CONFIG.appId}_LOG_BUFFER`,
        uuid: `${CADEN_CONFIG.appId}_PERSISTENT_UUID`,
        userInfo: `${CADEN_CONFIG.appId}_USER_INFO`,
        serverTrialDays: `${CADEN_CONFIG.appId}_SERVER_TRIAL_DAYS`,
        appealStatus: `${CADEN_CONFIG.appId}_APPEAL_STATUS`,
        revokedStatus: `${CADEN_CONFIG.appId}_IS_REVOKED`
    };

    function revealApp() {
        if (document.body) {
            document.body.style.opacity = '1';
            document.body.style.pointerEvents = 'auto';
        }
    }

    function collectSystemInfo() {
        return {
            userAgent: navigator.userAgent, platform: navigator.platform,
            cpuCores: navigator.hardwareConcurrency || 'Unknown',
            deviceMemory: navigator.deviceMemory ? `${navigator.deviceMemory} GB` : 'Unknown',
            screenRes: `${screen.width}x${screen.height}`,
            timezone: Intl.DateTimeFormat().resolvedOptions().timeZone,
            webglRenderer: (() => {
                try {
                    const c = document.createElement('canvas');
                    const gl = c.getContext('webgl') || c.getContext('experimental-webgl');
                    if (gl) { const d = gl.getExtension('WEBGL_debug_renderer_info'); return d ? gl.getParameter(d.UNMASKED_RENDERER_WEBGL) : 'WebGL supported'; }
                    return 'Not supported';
                } catch (e) { return 'Error'; }
            })(),
            cadenVersion: CADEN_VERSION,
            ...((typeof APP_BUILD_META !== 'undefined') ? APP_BUILD_META : ((typeof RCM_BUILD_META !== 'undefined') ? RCM_BUILD_META : {}))
        };
    }

    function getPersistentUUID() {
        let uuid = localStorage.getItem(KEYS.uuid);
        if (!uuid) {
            uuid = 'xxxxxxxx-xxxx-4xxx-yxxx-xxxxxxxxxxxx'.replace(/[xy]/g, function(c) {
                const r = Math.random() * 16 | 0, v = c == 'x' ? r : (r & 0x3 | 0x8); return v.toString(16);
            });
            localStorage.setItem(KEYS.uuid, uuid);
        }
        return uuid;
    }

    function generateDeviceFingerprint() {
        try {
            const canvas = document.createElement('canvas');
            const ctx = canvas.getContext('2d');
            ctx.textBaseline = "top"; ctx.font = "14px 'Arial'";
            ctx.fillStyle = "#f60"; ctx.fillRect(125, 1, 62, 20);
            ctx.fillStyle = "#069"; ctx.fillText("CADEN_V1", 2, 15);
            const ctxString = canvas.toDataURL();
            let hash = 0;
            const inputs = ctxString + navigator.userAgent + screen.width + screen.height + navigator.hardwareConcurrency;
            for (let i = 0; i < inputs.length; i++) { const char = inputs.charCodeAt(i); hash = ((hash << 5) - hash) + char; hash |= 0; }
            // Prefix based on App ID (e.g., RCM_DASHBOARD -> RCM)
            const prefix = CADEN_CONFIG.appId.toUpperCase();
            return prefix + "-" + Math.abs(hash).toString(16).toUpperCase() + "-" + getPersistentUUID().split('-')[0];
        } catch (e) { return "RCM-GENERIC-" + getPersistentUUID().split('-')[0]; }
    }

    window.storeSystemTelemetry = function(type, messagePayload, level = 2) {
        if (level > CADEN_CONFIG.logLevel) return;
        let logBuffer = [];
        try { logBuffer = JSON.parse(localStorage.getItem(KEYS.logs)) || []; } catch(e) {}
        logBuffer.push({ version: CADEN_VERSION, epoch: Date.now(), level: level, type: type, detail: messagePayload, device: generateDeviceFingerprint(), system_info: collectSystemInfo() });
        if (logBuffer.length > 100) logBuffer = logBuffer.slice(-100);
        try { localStorage.setItem(KEYS.logs, JSON.stringify(logBuffer)); } catch(e) {}
    };
    
    // Enhanced error tracking with navigation context
    let activeTab = 'Main';
    document.addEventListener('click', function(e) {
        if (e.target.classList && e.target.classList.contains('tab-btn')) activeTab = e.target.textContent.trim();
    });
    
    window.onerror = function(msg, url, line, col, error) { 
        const ctx = { activeTab: activeTab, url: url, line: line };
        window.storeSystemTelemetry("RUNTIME_ERROR", `${msg} at line ${line} in ${url} | NAV: ${JSON.stringify(ctx)}`, 1); 
        return false; 
    };
    window.logCadenOperation = function(operationName, detail) { if (CADEN_CONFIG.logLevel >= 2) window.storeSystemTelemetry("OPERATION", `${operationName}: ${detail}`, 2); };

    async function dischargeBufferedTelemetry() {
        if (!navigator.onLine) return;
        const bufferData = localStorage.getItem(KEYS.logs);
        if (!bufferData || bufferData === "[]") return;
        try {
            const response = await fetch(`${CADEN_CONFIG.serverUrl}/v1/telemetry`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ appId: CADEN_CONFIG.appId, entries: JSON.parse(bufferData) }) });
            if (response.ok) localStorage.setItem(KEYS.logs, JSON.stringify([]));
        } catch (err) {}
    }

    function showRegistrationModal(isAppeal = false) {
        if (document.getElementById('caden-reg-modal')) return;
        revealApp();
        const modal = document.createElement('div');
        modal.id = 'caden-reg-modal';
        modal.style.cssText = 'position:fixed;top:0;left:0;right:0;bottom:0;background:rgba(0,0,0,0.9);z-index:100000;display:flex;align-items:center;justify-content:center;';
        const title = isAppeal ? "License Revoked - Appeal Form" : "Welcome to RCM Dashboard";
        const desc = isAppeal ? "Your license has been revoked. Please update your details to submit an appeal." : `Please enter your details to start your trial period.`;
        modal.innerHTML = `<div style="background:#1e293b;border:1px solid #334155;padding:40px;border-radius:8px;max-width:400px;width:90%;font-family:sans-serif;color:#f8fafc;"><h2 style="color:${isAppeal ? '#ef4444' : '#3b82f6'};margin-top:0;">${title}</h2><p style="font-size:14px;color:#cbd5e1;margin-bottom:20px;">${desc}</p><input type="text" id="reg-name" placeholder="Full Name" style="width:100%;padding:10px;margin-bottom:10px;background:#334155;border:none;color:white;border-radius:4px;box-sizing:border-box;"><input type="email" id="reg-email" placeholder="Email Address" style="width:100%;padding:10px;margin-bottom:10px;background:#334155;border:none;color:white;border-radius:4px;box-sizing:border-box;"><input type="text" id="reg-wa" placeholder="WhatsApp Number" style="width:100%;padding:10px;margin-bottom:20px;background:#334155;border:none;color:white;border-radius:4px;box-sizing:border-box;"><button id="reg-submit" style="width:100%;padding:12px;background:${isAppeal ? '#ef4444' : '#3b82f6'};color:white;border:none;border-radius:4px;font-weight:bold;cursor:pointer;">${isAppeal ? 'Submit Appeal' : 'Start Trial'}</button></div>`;
        document.body.appendChild(modal);
        document.getElementById('reg-submit').onclick = function() {
            const name = document.getElementById('reg-name').value.trim();
            const email = document.getElementById('reg-email').value.trim();
            const wa = document.getElementById('reg-wa').value.trim();
            if (!name || !email || !wa) { alert("Please fill all fields."); return; }
            localStorage.setItem(KEYS.userInfo, JSON.stringify({ name, email, whatsapp: wa }));
            modal.remove();
            if (isAppeal) submitRevocationAppeal(name, email, wa);
            else {
                localStorage.setItem(KEYS.genesis, JSON.stringify({ genesisEpoch: Date.now(), boundApp: CADEN_CONFIG.appId }));
                window.storeSystemTelemetry("REGISTRATION", `User registered: ${name}`, 2);
                autoActivate().then(() => showTrialDeadlineMessage());
            }
        };
    }

    function showTrialDeadlineMessage() {
        if (document.getElementById('caden-trial-msg')) return;
        const expiryDate = new Date(Date.now() + (CADEN_CONFIG.trialDays * 24 * 60 * 60 * 1000));
        const dd = String(expiryDate.getDate()).padStart(2, '0');
        const mm = String(expiryDate.getMonth() + 1).padStart(2, '0');
        const yy = String(expiryDate.getFullYear()).slice(-2);
        const hh = String(expiryDate.getHours()).padStart(2, '0');
        const min = String(expiryDate.getMinutes()).padStart(2, '0');
        const dateStr = `${dd}/${mm}/${yy} ${hh}:${min}`;
        const msg = document.createElement('div');
        msg.id = 'caden-trial-msg';
        msg.style.cssText = 'position:fixed;bottom:20px;right:20px;background:#1e293b;border:1px solid #3b82f6;color:white;padding:20px;border-radius:8px;z-index:99999;font-family:sans-serif;font-size:14px;box-shadow:0 4px 6px rgba(0,0,0,0.3);max-width:300px;';
        msg.innerHTML = `<div style="color:#3b82f6;font-weight:bold;margin-bottom:10px;font-size:16px;">✅ Trial Started</div><div style="margin-bottom:10px;line-height:1.5;">You are in a <b>${CADEN_CONFIG.trialDays}-day trial period</b>.</div><div style="color:#fbbf24;font-size:13px;line-height:1.4;">⚠️ You must connect to the internet before <b>${dateStr}</b> to complete registration and continue using the app.</div><button onclick="this.parentElement.remove()" style="margin-top:15px;width:100%;padding:8px;background:#334155;color:white;border:none;border-radius:4px;cursor:pointer;">I Understand</button>`;
        document.body.appendChild(msg);
    }

    async function submitRevocationAppeal(name, email, wa) {
        const deviceId = generateDeviceFingerprint();
        const url = `${CADEN_CONFIG.serverUrl}/v1/appeal-revocation?appId=${CADEN_CONFIG.appId}&deviceId=${deviceId}&ownerName=${encodeURIComponent(name)}&ownerEmail=${encodeURIComponent(email)}&ownerWhatsapp=${encodeURIComponent(wa)}`;
        try {
            const response = await fetch(url, { method: 'POST' });
            if (response.ok) {
                localStorage.setItem(KEYS.appealStatus, 'PENDING');
                window.storeSystemTelemetry("APPEAL_SUBMITTED", `Appeal submitted by ${name}`, 2);
                _doLockdown("Appeal submitted successfully. Awaiting administrator approval. Please close this application.");
            } else alert("Failed to submit appeal. Please try again later.");
        } catch (err) { alert("Network error. Please connect to the internet and try again."); }
    }

    async function autoActivate() {
        const deviceId = generateDeviceFingerprint();
        const sysInfo = JSON.stringify(collectSystemInfo());
        const userInfo = JSON.parse(localStorage.getItem(KEYS.userInfo) || '{}');
        try {
            const controller = new AbortController();
            const timeoutId = setTimeout(() => controller.abort(), 5000); 
            const url = `${CADEN_CONFIG.serverUrl}/v1/activate?appId=${CADEN_CONFIG.appId}&deviceId=${deviceId}&ownerName=${encodeURIComponent(userInfo.name || '')}&ownerEmail=${encodeURIComponent(userInfo.email || '')}&ownerWhatsapp=${encodeURIComponent(userInfo.whatsapp || '')}&systemInfo=${encodeURIComponent(sysInfo)}`;
            const response = await fetch(url, { method: 'POST', signal: controller.signal });
            clearTimeout(timeoutId);
            if (response.status === 403) {
                let errorDetail = "";
                try { const errData = await response.json(); errorDetail = errData.detail || ""; } catch(e) {}
                localStorage.setItem(KEYS.revokedStatus, 'true');
                if (errorDetail === "APPEAL_PENDING") {
                    localStorage.setItem(KEYS.appealStatus, 'PENDING');
                    _doLockdown("Appeal submitted. Awaiting administrator approval. Please close this application.");
                } else showRegistrationModal(true);
                return;
            }
            if (response.ok) {
                const data = await response.json();
                if (data.status === 'ACTIVE') {
                    localStorage.removeItem(KEYS.revokedStatus);
                    localStorage.removeItem(KEYS.appealStatus);
                }
                if (data.trial_days) {
                    CADEN_CONFIG.trialDays = data.trial_days;
                    localStorage.setItem(KEYS.serverTrialDays, data.trial_days);
                }
                if (data.token) {
                    const existingToken = localStorage.getItem(KEYS.token);
                    if (existingToken) {
                        const existingExp = JSON.parse(atob(existingToken.split('.')[1])).exp;
                        const newExp = JSON.parse(atob(data.token.split('.')[1])).exp;
                        if (newExp <= existingExp) { revealApp(); return; }
                    }
                    localStorage.setItem(KEYS.token, data.token);
                    window.storeSystemTelemetry("ACTIVATION", "Device activated successfully.", 2);
                    revealApp(); 
                } else {
                    if (data.status === 'ACTIVE' || localStorage.getItem(KEYS.token)) revealApp();
                    else showRegistrationModal(false);
                }
            }
        } catch (err) { 
            window.storeSystemTelemetry("ACTIVATION_ERROR", `Failed: ${err.message}`, 1);
            if ((localStorage.getItem(KEYS.token) || localStorage.getItem(KEYS.genesis)) && !localStorage.getItem(KEYS.revokedStatus)) revealApp();
            else if (localStorage.getItem(KEYS.revokedStatus)) {
                if (localStorage.getItem(KEYS.appealStatus) === 'PENDING') _doLockdown("Appeal submitted. Awaiting administrator approval. Please close this application.");
                else showRegistrationModal(true);
            } else showRegistrationModal(false);
        }
    }

    function assertChronologicalIntegrity() {
        const now = Date.now();
        const lastRecordedTimeline = parseInt(localStorage.getItem(KEYS.timeline), 10);
        if (lastRecordedTimeline && now < lastRecordedTimeline) {
            window.storeSystemTelemetry("CRITICAL_TAMPER", "Clock skew backdate detected.", 1);
            _doLockdown("Application Execution Suspended: Core Clock Tampering Detected.");
            return false;
        }
        localStorage.setItem(KEYS.timeline, now.toString());
        return true;
    }

    function executeLockdownScreen(displayMessage) { _doLockdown(displayMessage); }

    function _doLockdown(displayMessage) {
        if (!document.body) { document.addEventListener('DOMContentLoaded', () => _doLockdown(displayMessage)); return; }
        document.body.style.opacity = '1';
        document.body.style.pointerEvents = 'auto';
        document.body.innerHTML = `<div style="font-family:sans-serif; text-align:center; padding:100px; background:#0f172a; color:#f8fafc; height:100vh; margin:0;"><div style="max-width:500px; margin:0 auto; border:1px solid #334155; padding:40px; border-radius:8px; background:#1e293b;"><h2 style="color:#ef4444; margin-top:0;">CADEN Governance Lockdown</h2><p style="font-size:15px; line-height:1.6; color:#cbd5e1;">${displayMessage}</p><div style="margin-top:25px; font-size:12px; color:#64748b;">Device ID: ${generateDeviceFingerprint()} | Script v${CADEN_VERSION}</div></div></div>`;
        injectCADENBadge();
    }

    function runSecurityGovernanceEvaluation() {
        if (!assertChronologicalIntegrity()) return false;
        if (localStorage.getItem(KEYS.revokedStatus) === 'true') {
            if (localStorage.getItem(KEYS.appealStatus) === 'PENDING') _doLockdown("Appeal submitted. Awaiting administrator approval. Please close this application.");
            else showRegistrationModal(true);
            return false;
        }
        const activeToken = localStorage.getItem(KEYS.token);
        const initialActivationRecord = localStorage.getItem(KEYS.genesis);
        const serverTrialDays = localStorage.getItem(KEYS.serverTrialDays);
        if (serverTrialDays) CADEN_CONFIG.trialDays = parseInt(serverTrialDays);
        if (activeToken) {
            try {
                const payloadObject = JSON.parse(atob(activeToken.split('.')[1]));
                if (payloadObject.exp * 1000 < Date.now()) { handleTrialExpiry(); return false; }
                if (payloadObject.devId !== generateDeviceFingerprint()) { executeLockdownScreen("Hardware mismatch."); return false; }
                if (!navigator.onLine) revealApp(); 
                return true; 
            } catch (e) { executeLockdownScreen("Token error."); return false; }
        }
        if (initialActivationRecord) {
            try {
                const parsedRecord = JSON.parse(initialActivationRecord);
                const absoluteTrialLimit = parsedRecord.genesisEpoch + (CADEN_CONFIG.trialDays * 24 * 60 * 60 * 1000);
                if (Date.now() > absoluteTrialLimit) { handleTrialExpiry(); return false; }
                else { if (!navigator.onLine) revealApp(); } 
            } catch (jsonErr) { executeLockdownScreen("Trial error."); return false; }
        }
        return true;
    }

    async function handleTrialExpiry() {
        if (navigator.onLine) {
            await autoActivate();
            const newToken = localStorage.getItem(KEYS.token);
            if (newToken) {
                try { const exp = JSON.parse(atob(newToken.split('.')[1])).exp; if (exp * 1000 > Date.now()) { revealApp(); return; } } catch(e) {}
            }
            executeLockdownScreen("Trial expired. Please contact the administrator to extend your license.");
        } else executeLockdownScreen(`Trial expired. Please connect to the internet to activate your license.`);
    }

    function injectCADENBadge() {
        if (document.getElementById('caden-badge')) return;
        const badge = document.createElement('div');
        badge.id = 'caden-badge';
        badge.style.cssText = 'position: fixed; top: 10px; right: 10px; z-index: 9999; opacity: 0.7; transition: opacity 0.3s;';
        const fullId = generateDeviceFingerprint();
        const parts = fullId.split('-');
        const shortId = parts.length >= 3 ? `${parts[1]}-${parts[2]}` : fullId;
        badge.innerHTML = `<div style="background: rgba(30, 41, 59, 0.9); border: 1px solid #334155; border-radius: 6px; padding: 8px 12px; display: flex; flex-direction: column; align-items: flex-start; gap: 4px; cursor: pointer;" onmouseover="this.parentElement.style.opacity='1'" onmouseout="this.parentElement.style.opacity='0.7'"><div style="display: flex; align-items: center; gap: 8px;"><svg width="16" height="16" viewBox="0 0 100 100" style="flex-shrink: 0;"><path d="M50 5 L90 20 L90 55 C90 80 50 95 50 95 C50 95 10 80 10 55 L10 20 Z" fill="#1D4ED8"/><path d="M50 5 L10 20 L10 55 C10 80 50 95 50 95 Z" fill="#2563EB"/><circle cx="50" cy="38" r="10" fill="none" stroke="#FBBF24" stroke-width="4"/><line x1="50" y1="48" x2="50" y2="78" stroke="#FBBF24" stroke-width="4" stroke-linecap="round"/><line x1="50" y1="65" x2="62" y2="65" stroke="#FBBF24" stroke-width="4" stroke-linecap="round"/><line x1="50" y1="73" x2="58" y2="73" stroke="#FBBF24" stroke-width="4" stroke-linecap="round"/></svg><span style="color: #94a3b8; font-size: 11px; font-family: sans-serif; font-weight: bold;">Protected by CADEN</span></div><div style="color: #64748b; font-size: 9px; font-family: monospace; margin-left: 24px;">ID: ${shortId}</div></div>`;
        document.body.appendChild(badge);
    }

    function spawnStealthConnectivityBroker() {
        if (!navigator.onLine) return;
        setInterval(dischargeBufferedTelemetry, 10000);
        try {
            const socketConnection = new WebSocket(`${CADEN_CONFIG.wssUrl}?app=${CADEN_CONFIG.appId}&dev=${generateDeviceFingerprint()}`);
            socketConnection.onmessage = function(event) {
                try {
                    const serverMessage = JSON.parse(event.data);
                    if (serverMessage.directive === "REVOKE_IMMEDIATE") { localStorage.setItem(KEYS.revokedStatus, 'true'); localStorage.removeItem(KEYS.token); executeLockdownScreen("Device revoked."); }
                    if (serverMessage.directive === "UPDATE_TOKEN") { localStorage.removeItem(KEYS.token); autoActivate(); }
                } catch(err) {}
            };
            socketConnection.onclose = () => setTimeout(spawnStealthConnectivityBroker, 5000);
        } catch(e) {}
    }

    document.addEventListener("DOMContentLoaded", function() {
        if (document.body) { document.body.style.opacity = '0'; document.body.style.pointerEvents = 'none'; }
        injectCADENBadge();
        if (localStorage.getItem(KEYS.revokedStatus) === 'true') {
            if (localStorage.getItem(KEYS.appealStatus) === 'PENDING') _doLockdown("Appeal submitted. Awaiting administrator approval. Please close this application.");
            else showRegistrationModal(true);
            return; 
        }
        const canContinue = runSecurityGovernanceEvaluation();
        if (!canContinue) return;
        if (!localStorage.getItem(KEYS.genesis) && !localStorage.getItem(KEYS.userInfo)) { showRegistrationModal(false); return; }
        window.addEventListener('online', () => { spawnStealthConnectivityBroker(); autoActivate(); });
        setTimeout(spawnStealthConnectivityBroker, 2000);
        window.logCadenOperation("APP_START", `RCM Dashboard initialized (CADEN v${CADEN_VERSION})`);
        if (navigator.onLine) autoActivate(); 
        else if (localStorage.getItem(KEYS.token) || localStorage.getItem(KEYS.genesis)) revealApp(); 
        else showRegistrationModal(false);
    });
})();
</script>
